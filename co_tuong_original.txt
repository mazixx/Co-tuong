def inside(r, c):
    return 0 <= r < 10 and 0 <= c < 9


def palace(r, c, side):
    return (
        3 <= c <= 5
        and (7 <= r <= 9 if side == RED else 0 <= r <= 2)
    )


def crossed(r, side):
    return r <= 4 if side == RED else r >= 5


def find_king(board, side):
    for r in range(10):
        for c in range(9):
            if board[r][c] == side * KING:
                return r, c
    return None


def targets(board, r, c):
    """Cách đi của quân, chưa kiểm tra tự chiếu."""
    piece = board[r][c]
    if not piece:
        return []

    side = RED if piece > 0 else BLACK
    kind = abs(piece)
    result = []

    def add(nr, nc):
        if inside(nr, nc) and board[nr][nc] * side <= 0:
            result.append((nr, nc))

    if kind == KING:
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if palace(nr, nc, side):
                add(nr, nc)

        # Tướng đối mặt.
        for dr in (-1, 1):
            nr = r + dr
            while inside(nr, c):
                if board[nr][c]:
                    if board[nr][c] == -side * KING:
                        result.append((nr, c))
                    break
                nr += dr

    elif kind == ADVISOR:
        for dr, dc in ((1, 1), (1, -1), (-1, 1), (-1, -1)):
            nr, nc = r + dr, c + dc
            if palace(nr, nc, side):
                add(nr, nc)

    elif kind == ELEPHANT:
        for dr, dc in ((2, 2), (2, -2), (-2, 2), (-2, -2)):
            nr, nc = r + dr, c + dc
            if not inside(nr, nc):
                continue
            if side == RED and nr < 5:
                continue
            if side == BLACK and nr > 4:
                continue
            if board[r + dr // 2][c + dc // 2] == 0:
                add(nr, nc)

    elif kind == HORSE:
        steps = [
            (-2, -1, -1, 0), (-2, 1, -1, 0),
            (2, -1, 1, 0), (2, 1, 1, 0),
            (-1, -2, 0, -1), (1, -2, 0, -1),
            (-1, 2, 0, 1), (1, 2, 0, 1),
        ]
        for dr, dc, lr, lc in steps:
            nr, nc = r + dr, c + dc
            if inside(nr, nc) and board[r + lr][c + lc] == 0:
                add(nr, nc)

    elif kind in (ROOK, CANNON):
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            screen_found = False

            while inside(nr, nc):
                target = board[nr][nc]

                if kind == ROOK:
                    if target == 0:
                        result.append((nr, nc))
                    else:
                        if target * side < 0:
                            result.append((nr, nc))
                        break
                else:
                    if not screen_found:
                        if target == 0:
                            result.append((nr, nc))
                        else:
                            screen_found = True
                    elif target:
                        if target * side < 0:
                            result.append((nr, nc))
                        break

                nr += dr
                nc += dc

    elif kind == PAWN:
        add(r - 1 if side == RED else r + 1, c)
        if crossed(r, side):
            add(r, c - 1)
            add(r, c + 1)

    return result


def in_check(board, side):
    king = find_king(board, side)
    if king is None:
        return True

    for r in range(10):
        for c in range(9):
            if board[r][c] * side < 0:
                if king in targets(board, r, c):
                    return True
    return False


def make_move(board, move):
    r1, c1, r2, c2 = move
    captured = board[r2][c2]
    board[r2][c2] = board[r1][c1]
    board[r1][c1] = 0
    return captured


def unmake_move(board, move, captured):
    r1, c1, r2, c2 = move
    board[r1][c1] = board[r2][c2]
    board[r2][c2] = captured


def position_key(board, side):
    return tuple(v for row in board for v in row), side


def legal_moves(board, side, counts=None):
    """
    counts: lịch sử số lần xuất hiện thế cờ.
    Nếu có counts, cấm nước tạo thế lặp lần thứ ba.
    """
    if find_king(board, side) is None:
        return []

    moves = []

    for r in range(10):
        for c in range(9):
            if board[r][c] * side <= 0:
                continue

            for nr, nc in targets(board, r, c):
                move = (r, c, nr, nc)
                captured = make_move(board, move)

                valid = not in_check(board, side)

                if valid and counts is not None:
                    key = position_key(board, -side)
                    valid = counts.get(key, 0) < 2

                unmake_move(board, move, captured)

                if valid:
                    moves.append(move)

    return moves


def move_name(board, move):
    r1, c1, r2, c2 = move
    name = VI[abs(board[r1][c1])]
    action = "ăn" if board[r2][c2] else "→"

    return (
        f"{name} {chr(65 + c1)}{r1} "
        f"{action} {chr(65 + c2)}{r2}"
    )


def evaluate(board, side=RED):
    total = 0

    for r in range(10):
        for c in range(9):
            piece = board[r][c]
            if not piece:
                continue

            owner = RED if piece > 0 else BLACK
            kind = abs(piece)
            value = VALUES[kind]

            advance = 9 - r if owner == RED else r
            center = 4 - abs(c - 4)

            if kind == PAWN:
                value += advance * 8
                if crossed(r, owner):
                    value += 70 + center * 10

            elif kind == HORSE:
                value += center * 12 + min(advance, 6) * 4
                if r in (0, 9):
                    value -= 20

            elif kind == CANNON:
                value += center * 7 + min(advance, 5) * 3

            elif kind == ROOK:
                value += center * 4 + min(advance, 7) * 3

            total += owner * value

    return total * side

class SearchStopped(Exception):
    pass


class Engine:
    def __init__(self, seconds, cancel):
        self.deadline = time.perf_counter() + seconds
        self.cancel = cancel
        self.nodes = 0

    def check(self):
        if self.cancel.is_set() or time.perf_counter() >= self.deadline:
            raise SearchStopped()

    def ordered(self, board, moves, preferred=None):
        def priority(move):
            r1, c1, r2, c2 = move
            victim = board[r2][c2]
            attacker = board[r1][c1]

            value = 0
            if victim:
                value += 10000 + VALUES[abs(victim)] * 10
                value -= VALUES[abs(attacker)]
            if move == preferred:
                value += 10000000
            return value

        return sorted(moves, key=priority, reverse=True)

    def search_node(self, board, side, depth, alpha, beta,
                    ply, counts, quiet_left=1):
        self.check()
        self.nodes += 1

        moves = legal_moves(board, side, counts)

        if not moves:
            return -MATE + ply

        if depth <= 0:
            score = evaluate(board, side)

            if quiet_left <= 0:
                return score

            if not in_check(board, side):
                if score >= beta:
                    return score
                alpha = max(alpha, score)
                moves = [m for m in moves if board[m[2]][m[3]]]
                if not moves:
                    return alpha

        for move in self.ordered(board, moves):
            self.check()
            captured = make_move(board, move)
            key = position_key(board, -side)
            counts[key] += 1

            try:
                value = -self.search_node(
                    board, -side, depth - 1,
                    -beta, -alpha, ply + 1, counts,
                    quiet_left - 1 if depth <= 0 else quiet_left
                )
            finally:
                counts[key] -= 1
                if counts[key] == 0:
                    del counts[key]
                unmake_move(board, move, captured)

            if value >= beta:
                return value

            alpha = max(alpha, value)

        return alpha

    def search(self, board, side, max_depth, counts, exact=False):
        counts = counts.copy()
        moves = legal_moves(board, side, counts)

        if not moves:
            return None, None, 0, {}

        best_move = self.ordered(board, moves)[0]
        best_score = None
        completed_depth = 0
        completed_scores = {}

        for depth in range(1, max_depth + 1):
            scores = {}
            alpha = -INF

            try:
                for move in self.ordered(board, moves, best_move):
                    self.check()
                    captured = make_move(board, move)
                    key = position_key(board, -side)
                    counts[key] += 1

                    try:
                        score = -self.search_node(
                            board, -side, depth - 1,
                            -INF, INF if exact else -alpha,
                            1, counts
                        )
                    finally:
                        counts[key] -= 1
                        if counts[key] == 0:
                            del counts[key]
                        unmake_move(board, move, captured)

                    scores[move] = score
                    alpha = max(alpha, score)

            except SearchStopped:
                break

            best_move = max(scores, key=scores.get)
            best_score = scores[best_move]
            completed_depth = depth
            completed_scores = scores

            if best_score > MATE - 100:
                break

        return best_move, best_score, completed_depth, completed_scores


def grade_move(before, move, counts, cancel):
    engine = Engine(0.8, cancel)
    _, score, depth, scores = engine.search(
        before, RED, 2, counts, exact=True
    )

    if score is None or move not in scores:
        return "Chưa đủ thời gian để chấm nước."

    loss = max(0, score - scores[move])

    if loss <= 25:
        name = "Gần tốt nhất"
    elif loss <= 90:
        name = "Khá tốt"
    elif loss <= 200:
        name = "Chưa chính xác"
    elif loss <= 400:
        name = "Sai sót"
    else:
        name = "Sai lầm lớn"

    if loss > MATE // 2:
        return f"{name} · bỏ lỡ hoặc rơi vào biến quyết định"

    return f"{name} · hụt ~{loss / 100:.1f} binh · d{depth}"

def mix(a, b, t):
    return tuple(int(a[i] + (b[i] - a[i]) * t) for i in range(3))


def text(message, x, y, color=TEXT, font=FONT, center=False):
    image = font.render(str(message), True, color)
    rect = image.get_rect()
    if center:
        rect.center = (x, y)
    else:
        rect.topleft = (x, y)
    screen.blit(image, rect)


def wrapped(message, x, y, width, color=TEXT, font=SMALL,
            line_h=22, max_lines=3):
    lines = []
    current = ""

    for word in str(message).split():
        candidate = f"{current} {word}".strip()
        if font.size(candidate)[0] > width and current:
            lines.append(current)
            current = word
        else:
            current = candidate

    if current:
        lines.append(current)

    if len(lines) > max_lines:
        lines = lines[:max_lines]
        lines[-1] += "..."

    for line in lines:
        text(line, x, y, color, font)
        y += line_h

    return y

def board_xy(r, c):
    return OX + c * CELL, OY + r * CELL


def mouse_square(pos):
    x, y = pos
    c = round((x - OX) / CELL)
    r = round((y - OY) / CELL)

    if not inside(r, c):
        return None

    px, py = board_xy(r, c)
    if math.hypot(x - px, y - py) <= 28:
        return r, c
    return None


def time_string(seconds):
    seconds = max(0, math.ceil(seconds))
    return f"{seconds // 60:02d}:{seconds % 60:02d}"


def draw_arrow(move):
    r1, c1, r2, c2 = move
    start = pygame.Vector2(board_xy(r1, c1))
    end = pygame.Vector2(board_xy(r2, c2))
    direction = end - start

    if direction.length() == 0:
        return

    direction = direction.normalize()
    start += direction * 27
    end -= direction * 14
    perpendicular = pygame.Vector2(-direction.y, direction.x)

    color = (29, 146, 83)
    pygame.draw.line(screen, color, start, end, 6)
    pygame.draw.polygon(
        screen, color,
        [
            end,
            end - direction * 18 + perpendicular * 10,
            end - direction * 18 - perpendicular * 10,
        ]
    )

class Game:
    def __init__(self):
        self.level = 0
        self.generation = 0
        self.cancel = threading.Event()
        self.results = queue.Queue()

        self.background_index = random.randrange(len(THEMES))
        self.background_cache = {}
        self.load_background()

        self.new_game()

    def load_background(self):
        index = self.background_index
        if index not in self.background_cache:
            self.background_cache[index] = make_background(index)
        self.background = self.background_cache[index]

    def change_background(self):
        choices = [
            i for i in range(len(THEMES))
            if i != self.background_index
        ]
        self.background_index = random.choice(choices)
        self.load_background()

    def invalidate_jobs(self):
        self.cancel.set()
        self.cancel = threading.Event()
        self.generation += 1

    def new_game(self):
        self.invalidate_jobs()

        self.board = initial_board()
        self.turn = RED
        self.remaining = {RED: TOTAL_TIME, BLACK: TOTAL_TIME}

        self.finished = False
        self.paused = False
        self.pause_started = None
        self.thinking = False

        self.selected = None
        self.last_move = None
        self.history = []
        self.history_scroll = 0

        self.undo_stack = []
        self.undo_accepted = 0
        self.undo_requested = set()
        self.draw_requested = set()

        self.pending_ai = None
        self.hint = None
        self.hint_requested = False
        self.hint_status = "Gợi ý tự xuất hiện sau 30 giây suy nghĩ."

        self.message = "Lượt của bạn — quân Đỏ."
        self.note = "Mỗi bên 10 phút. Xin đi lại tối đa 2 lần được chấp nhận."
        self.ai_info = "AI chưa phân tích."

        self.counts = Counter()
        self.counts[position_key(self.board, RED)] = 1
        self.refresh_moves()

        now = time.perf_counter()
        self.last_clock = now
        self.turn_started = now

    def refresh_moves(self):
        self.basic_moves = legal_moves(self.board, self.turn)
        self.moves = legal_moves(self.board, self.turn, self.counts)
        self.blocked_moves = set(self.basic_moves) - set(self.moves)
        self.checked = in_check(self.board, self.turn)

    def finish(self, message):
        self.finished = True
        self.thinking = False
        self.selected = None
        self.message = message
        self.invalidate_jobs()

    def charge_clock(self):
        now = time.perf_counter()

        if self.paused or self.finished:
            self.last_clock = now
            return

        elapsed = now - self.last_clock
        self.last_clock = now
        self.remaining[self.turn] = max(
            0, self.remaining[self.turn] - elapsed
        )

        if self.remaining[self.turn] <= 0:
            self.finish(
                "Bạn hết giờ — máy thắng."
                if self.turn == RED
                else "Máy hết giờ — bạn thắng!"
            )

    def toggle_pause(self):
        if self.finished:
            return

        if not self.paused:
            self.charge_clock()
            if self.finished:
                return

            self.paused = True
            self.pause_started = time.perf_counter()
            self.saved_message = self.message
            self.message = "Ván đấu đang tạm dừng."
            self.thinking = False

            self.invalidate_jobs()

            if self.hint is None:
                self.hint_requested = False

        else:
            now = time.perf_counter()
            self.turn_started += now - self.pause_started
            self.last_clock = now

            self.paused = False
            self.pause_started = None
            self.message = self.saved_message

            if self.turn == BLACK:
                self.start_ai()

    def start_ai(self):
        if self.finished or self.paused:
            return

        self.thinking = True
        token = self.generation
        cancel = self.cancel
        results = self.results

        board = [row[:] for row in self.board]
        counts = self.counts.copy()
        level = self.level
        pending = self.pending_ai

        def worker():
            try:
                if pending is not None:
                    before, move, previous_counts, index = pending
                    comment = grade_move(
                        [row[:] for row in before],
                        move, previous_counts, cancel
                    )

                    if cancel.is_set():
                        return
                    results.put((token, "grade", (index, comment)))

                _, depth_limit, seconds = LEVELS[level]
                engine = Engine(seconds, cancel)
                move, score, depth, _ = engine.search(
                    board, BLACK, depth_limit, counts
                )

                if not cancel.is_set():
                    results.put(
                        (token, "ai", (move, score, depth, engine.nodes))
                    )

            except Exception as exc:
                if not cancel.is_set():
                    results.put((token, "error", str(exc)))

        threading.Thread(target=worker, daemon=True).start()

    def start_hint(self):
        self.hint_requested = True
        self.hint_status = "Đã suy nghĩ 30 giây: đang phân tích gợi ý..."

        token = self.generation
        cancel = self.cancel
        results = self.results

        board = [row[:] for row in self.board]
        counts = self.counts.copy()

        def worker():
            try:
                engine = Engine(3.0, cancel)
                move, score, depth, _ = engine.search(
                    board, RED, 5, counts
                )

                if not cancel.is_set():
                    results.put((token, "hint", (move, depth)))

            except Exception:
                if not cancel.is_set():
                    results.put((token, "hint_error", None))

        threading.Thread(target=worker, daemon=True).start()

    def save_position(self):
        return {
            "board": [row[:] for row in self.board],
            "history": [entry.copy() for entry in self.history],
            "counts": self.counts.copy(),
            "last_move": self.last_move,
        }

    def play_move(self, move, ai_depth=0):
        self.charge_clock()

        if self.finished or self.paused or move not in self.moves:
            return

        side = self.turn
        name = move_name(self.board, move)

        if side == RED:
            self.undo_stack.append(self.save_position())

            self.pending_ai = (
                [row[:] for row in self.board],
                move,
                self.counts.copy(),
                len(self.history)
            )

        self.invalidate_jobs()
        make_move(self.board, move)

        self.history.append({
            "side": side,
            "name": name,
            "comment": (
                "Đang chấm nước..."
                if side == RED
                else f"AI chọn · độ sâu hoàn thành {ai_depth}"
            ),
        })

        self.last_move = move
        self.selected = None
        self.history_scroll = 0

        self.hint = None
        self.hint_requested = False
        self.hint_status = "Gợi ý tự xuất hiện sau 30 giây suy nghĩ."

        self.turn = -side
        self.thinking = False

        if self.turn == RED:
            self.pending_ai = None

        key = position_key(self.board, self.turn)
        self.counts[key] += 1

        now = time.perf_counter()
        self.last_clock = now
        self.turn_started = now

        self.refresh_moves()

        if not self.moves:
            self.history[-1]["comment"] = "Nước kết thúc ván."

            if self.basic_moves:
                reason = "chỉ còn nước bị cấm lặp lần thứ ba"
            else:
                reason = "hết nước hợp lệ"

            self.finish(
                f"Bạn thắng! Đen {reason}."
                if self.turn == BLACK
                else f"Máy thắng! Đỏ {reason}."
            )
            return

        if self.turn == BLACK:
            self.message = (
                "Đen bị chiếu — máy đang nghĩ..."
                if self.checked
                else "Máy đang suy nghĩ..."
            )
            self.start_ai()
        else:
            self.message = (
                "Bạn đang bị chiếu!"
                if self.checked
                else "Lượt của bạn — quân Đỏ."
            )

    def can_request_undo(self):
        return (
            not self.finished
            and not self.paused
            and self.turn == RED
            and bool(self.undo_stack)
            and len(self.history) >= 2
            and self.undo_accepted < MAX_UNDO
            and position_key(self.board, self.turn) not in self.undo_requested
        )

    def request_undo(self):
        self.charge_clock()

        if not self.can_request_undo():
            return

        key = position_key(self.board, self.turn)
        self.undo_requested.add(key)

        black_advantage = evaluate(self.board, BLACK)

        if self.checked:
            self.note = "Máy từ chối đi lại: bạn đang bị chiếu."
            return

        if black_advantage > 180:
            self.note = "Máy từ chối đi lại: đang có lợi thế rõ rệt."
            return

        if self.remaining[RED] < 20:
            self.note = "Máy từ chối đi lại: bạn còn quá ít thời gian."
            return

        snapshot = self.undo_stack.pop()

        self.invalidate_jobs()
        self.board = [row[:] for row in snapshot["board"]]
        self.history = [entry.copy() for entry in snapshot["history"]]
        self.counts = snapshot["counts"].copy()
        self.last_move = snapshot["last_move"]

        self.turn = RED
        self.selected = None
        self.thinking = False
        self.pending_ai = None
        self.history_scroll = 0

        self.hint = None
        self.hint_requested = False
        self.hint_status = "Gợi ý tự xuất hiện sau 30 giây suy nghĩ."

        self.undo_accepted += 1
        self.refresh_moves()

        now = time.perf_counter()
        self.last_clock = now
        self.turn_started = now

        self.message = "Máy đồng ý — bạn hãy chọn nước khác."
        self.note = (
            f"Đã thu hồi một cặp nước. "
            f"Còn {MAX_UNDO - self.undo_accepted} lần; không hoàn thời gian."
        )
        self.ai_info = "Thế cờ đã được khôi phục."

    def offer_draw(self):
        self.charge_clock()

        if self.finished or self.paused or self.turn != RED:
            return

        key = position_key(self.board, self.turn)
        if key in self.draw_requested:
            self.note = "Bạn đã cầu hòa tại thế cờ này."
            return

        self.draw_requested.add(key)

        if len(self.history) < 12:
            self.note = "Máy từ chối cầu hòa: ván đấu còn quá sớm."
            return

        if self.remaining[RED] < 30 and self.remaining[BLACK] > 60:
            self.note = "Máy từ chối: đang có lợi thế thời gian."
            return

        advantage = evaluate(self.board, BLACK)
        attackers = sum(
            abs(p) in (HORSE, ROOK, CANNON, PAWN)
            for row in self.board for p in row
        )

        accept = (
            advantage < -180
            or (abs(advantage) <= 100 and attackers <= 6)
            or (self.remaining[BLACK] < 45 and advantage < 100)
        )

        if accept:
            self.finish("Hòa! Máy đồng ý đề nghị cầu hòa.")
            self.note = "Đánh giá dựa trên thế cờ và thời gian."
        else:
            self.note = "Máy từ chối cầu hòa: muốn tiếp tục thi đấu."

    def resign(self):
        self.charge_clock()

        if not self.finished and not self.paused:
            self.finish("Bạn đã xin thua — máy thắng.")
            self.note = "Nhấn Ván mới để bắt đầu lại."

    def update(self):
        self.charge_clock()

        if self.paused:
            return

        while True:
            try:
                token, kind, data = self.results.get_nowait()
            except queue.Empty:
                break

            if token != self.generation or self.finished:
                continue

            if kind == "grade":
                index, comment = data
                if index < len(self.history):
                    self.history[index]["comment"] = comment

            elif kind == "ai":
                move, score, depth, nodes = data

                if self.turn != BLACK:
                    continue

                if score is None:
                    self.ai_info = f"AI: d{depth} · {nodes:,} nút."
                elif abs(score) > MATE // 2:
                    self.ai_info = "Engine thấy biến thắng/thua quyết định."
                else:
                    self.ai_info = (
                        f"Phân tích gần nhất: Đỏ {-score / 100:+.1f}"
                        f" · d{depth} · {nodes:,} nút"
                    )

                if move in self.moves:
                    self.play_move(move, depth)
                else:
                    self.finish("Engine không trả về nước hợp lệ.")

            elif kind == "hint":
                move, depth = data

                if self.turn == RED and move in self.moves:
                    if depth > 0:
                        self.hint = move
                        self.hint_status = (
                            f"Gợi ý: {move_name(self.board, move)}. "
                            f"Độ sâu {depth}; xem mũi tên xanh."
                        )
                    else:
                        self.hint_status = (
                            "Chưa đủ thời gian để phân tích gợi ý."
                        )

            elif kind == "hint_error":
                self.hint_status = "Không phân tích được gợi ý lượt này."

            elif kind == "error":
                self.finish("Engine gặp lỗi — hãy tạo ván mới.")
                self.note = str(data)[:110]

        if (
            not self.finished
            and self.turn == RED
            and not self.hint_requested
            and time.perf_counter() - self.turn_started >= HINT_AFTER
        ):
            self.start_hint()

    def click(self, pos):
        self.charge_clock()

        if BG_RECT.collidepoint(pos):
            self.change_background()
            return

        if NEW_RECT.collidepoint(pos):
            self.new_game()
            return

        for i, rect in enumerate(LEVEL_RECTS):
            if rect.collidepoint(pos):
                self.level = i
                self.new_game()
                return

        if PAUSE_RECT.collidepoint(pos):
            self.toggle_pause()
            return

        if self.finished or self.paused:
            return

        if UNDO_RECT.collidepoint(pos):
            self.request_undo()
            return

        if DRAW_RECT.collidepoint(pos):
            self.offer_draw()
            return

        if RESIGN_RECT.collidepoint(pos):
            self.resign()
            return

        if self.turn != RED:
            return

        square = mouse_square(pos)
        if square is None:
            self.selected = None
            return

        r, c = square

        if self.selected is not None:
            sr, sc = self.selected
            move = (sr, sc, r, c)

            if move in self.moves:
                self.play_move(move)
                return

            if move in self.blocked_moves:
                self.note = (
                    "Nước này bị khóa: tạo cùng thế cờ lần thứ ba. "
                    "Hãy chọn nước khác."
                )
                return

        self.selected = square if self.board[r][c] > 0 else None

    def scroll_history(self, amount):
        limit = max(0, len(self.history) - 4)
        self.history_scroll = max(
            0, min(limit, self.history_scroll + amount)
        )

            pygame.draw.circle(
                screen, (28, 143, 90), board_xy(sr, sc), 29, 3
            )

            for r1, c1, r2, c2 in self.moves:
                if (r1, c1) != self.selected:
                    continue

                point = board_xy(r2, c2)

                if self.board[r2][c2]:
                    pygame.draw.circle(
                        screen, (28, 143, 90), point, 29, 3
                    )
                else:
                    pygame.draw.circle(
                        screen, (28, 143, 90), point, 7
                    )

            # Dấu X: nước bị cấm vì lặp lần thứ ba.
            for r1, c1, r2, c2 in self.blocked_moves:
                if (r1, c1) != self.selected:
                    continue

                x, y = board_xy(r2, c2)
                pygame.draw.line(
                    screen, (190, 62, 40),
                    (x - 9, y - 9), (x + 9, y + 9), 3
                )
                pygame.draw.line(
                    screen, (190, 62, 40),
                    (x + 9, y - 9), (x - 9, y + 9), 3
                )

        if self.paused:
            cover = pygame.Surface((560, 610), pygame.SRCALPHA)
            cover.fill((6, 23, 19, 205))
            screen.blit(cover, (25, 85))

            text(
                "TẠM DỪNG", 305, 340,
                GOLD, TITLE, center=True
            )
            text(
                "Đồng hồ và AI đã dừng",
                305, 389, CREAM, FONT, center=True
            )
            text(
                "Nhấn Space hoặc Tiếp tục",
                305, 426, MUTED, SMALL, center=True
            )

def main():
    game = Game()
    clock = pygame.time.Clock()
    running = True

    while running:
        clock.tick(60)
        game.update()

        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

            elif event.type == pygame.MOUSEBUTTONDOWN:
                if event.button == 1:
                    game.click(event.pos)

            elif event.type == pygame.MOUSEWHEEL:
                if HISTORY_RECT.collidepoint(pygame.mouse.get_pos()):
                    game.scroll_history(event.y)

            elif event.type == pygame.KEYDOWN:
                if event.key == pygame.K_ESCAPE:
                    running = False

                elif event.key == pygame.K_SPACE:
                    game.toggle_pause()

                elif event.key == pygame.K_b:
                    game.change_background()

                elif event.key == pygame.K_n:
                    game.new_game()

                elif event.key in (pygame.K_1, pygame.K_2, pygame.K_3):
                    game.level = {
                        pygame.K_1: 0,
                        pygame.K_2: 1,
                        pygame.K_3: 2,
                    }[event.key]
                    game.new_game()

        game.draw()
        pygame.display.flip()

    game.cancel.set()
    pygame.quit()


if __name__ == "__main__":
    main()
